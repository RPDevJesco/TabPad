cmake_minimum_required(VERSION 3.16)
project(tabpad C)

set(TS vendor/tree-sitter/lib)

if(MSVC)
    set(FREE /GS-)
    set(WARN /W4)
    set(QUIET /w)
else()
    set(FREE -ffreestanding -fno-builtin -fno-stack-protector)
    set(WARN -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes -Wmissing-prototypes)
    set(QUIET -w)
endif()

if(WIN32)
    set(STORE port/win32/store_win32.c)
    set(DESK port/win32/desk_win32.c port/win32/tabpad.rc)
    enable_language(RC)
else()
    set(STORE port/posix/store_posix.c)
    set(DESK port/posix/desk_posix.c)
endif()

add_library(hl_own OBJECT src/hl.c src/hl_text.c src/hl_re.c src/hl_query.c src/hl_libc.c)
set_target_properties(hl_own PROPERTIES C_STANDARD 99 C_EXTENSIONS OFF)
target_compile_options(hl_own PRIVATE ${WARN} ${FREE})
target_include_directories(hl_own PRIVATE shim include ${TS}/include)
add_library(hl_ts OBJECT ${TS}/src/lib.c)
set_target_properties(hl_ts PROPERTIES C_STANDARD 11 C_EXTENSIONS OFF)
target_compile_options(hl_ts PRIVATE ${QUIET} ${FREE})
target_compile_definitions(hl_ts PRIVATE NDEBUG)
target_include_directories(hl_ts PRIVATE shim ${TS}/include ${TS}/src)
add_library(hl STATIC $<TARGET_OBJECTS:hl_own> $<TARGET_OBJECTS:hl_ts>)
target_include_directories(hl PUBLIC include)
add_library(sess STATIC src/sess.c src/sess_crc.c)
set_target_properties(sess PROPERTIES C_STANDARD 99 C_EXTENSIONS OFF)
target_compile_options(sess PRIVATE ${WARN} ${FREE})
target_include_directories(sess PUBLIC include)
add_library(ed STATIC src/ed.c src/ed_tab.c src/ed_edit.c src/ed_view.c src/ed_file.c src/ed_undo.c
    src/ed_cmd.c src/ed_menu.c src/ed_tool.c src/ed_find.c src/ed_status.c src/ed_word.c src/ed_cfg.c src/ed_list.c)
set_target_properties(ed PROPERTIES C_STANDARD 99 C_EXTENSIONS OFF)
target_compile_options(ed PRIVATE ${WARN} ${FREE})
target_include_directories(ed PUBLIC include)
add_executable(embed tools/embed.c)
set(LANG_NAMES c cpp csharp css javascript html lua apex python ruby rust markdown markdowninline json xml bash powershell)
set(SCAN_NAMES cpp csharp css javascript html lua python ruby rust markdown markdowninline xml bash powershell)
set(V ${CMAKE_CURRENT_SOURCE_DIR}/vendor/tree-sitter)
set(L ${CMAKE_CURRENT_SOURCE_DIR}/lang)
set(Q_cpp ${V}-c/queries/highlights.scm ${V}-cpp/queries/highlights.scm)
set(Q_javascript ${V}-javascript/queries/highlights.scm ${V}-javascript/queries/highlights-jsx.scm
    ${V}-javascript/queries/highlights-params.scm ${L}/javascript.scm)
set(Q_apex ${V}-apex/queries/highlights.scm ${V}-apex/queries/highlights-sosl.scm ${V}-apex/queries/highlights-soql.scm)
set(Q_json ${V}-json/queries/highlights.scm ${L}/json.scm)
set(Q_lua ${V}-lua/queries/highlights.scm ${L}/lua.scm)
set(Q_python ${V}-python/queries/highlights.scm ${L}/python.scm)
set(Q_powershell ${V}-powershell/queries/highlights.scm ${L}/powershell.scm)

set(LANG_SOURCES)

foreach(name ${LANG_NAMES})
    if(NOT DEFINED Q_${name})
        set(Q_${name} ${V}-${name}/queries/highlights.scm)
    endif()

    add_custom_command(
        OUTPUT query_${name}.c
        COMMAND embed hl_query_${name} query_${name}.c ${Q_${name}}
        DEPENDS embed ${Q_${name}})
    list(APPEND LANG_SOURCES vendor/tree-sitter-${name}/src/parser.c ${CMAKE_CURRENT_BINARY_DIR}/query_${name}.c)
endforeach()

foreach(name ${SCAN_NAMES})
    list(APPEND LANG_SOURCES vendor/tree-sitter-${name}/src/scanner.c)
endforeach()

add_library(hl_langs STATIC ${LANG_SOURCES})
set_target_properties(hl_langs PROPERTIES C_STANDARD 11 C_EXTENSIONS OFF)
target_compile_options(hl_langs PRIVATE ${QUIET} ${FREE})
target_compile_definitions(hl_langs PRIVATE NDEBUG)
target_include_directories(hl_langs PRIVATE shim vendor/tree-sitter-xml/src)

add_library(port_hosted OBJECT port/hosted/port_hosted.c)
add_library(store OBJECT ${STORE})
add_executable(test_hl test/test_hl.c $<TARGET_OBJECTS:port_hosted>)
add_executable(test_langs test/test_langs.c lang/langs.c port/theme_dark.c $<TARGET_OBJECTS:port_hosted>)
add_executable(test_sess test/test_sess.c $<TARGET_OBJECTS:port_hosted> $<TARGET_OBJECTS:store>)
add_executable(hlcat tools/hlcat.c lang/langs.c $<TARGET_OBJECTS:port_hosted>)
add_library(editor OBJECT lang/langs.c port/theme_dark.c port/icons.c)
set(EDITOR $<TARGET_OBJECTS:editor> $<TARGET_OBJECTS:port_hosted> $<TARGET_OBJECTS:store>)
add_executable(test_ed test/test_ed.c ${EDITOR})

foreach(hosted port_hosted store editor test_hl test_langs test_sess test_ed hlcat)
    set_target_properties(${hosted} PROPERTIES C_STANDARD 99 C_EXTENSIONS OFF)
    target_compile_options(${hosted} PRIVATE ${WARN})
    target_include_directories(${hosted} PRIVATE include src lang port port/hosted)
endforeach()

target_link_libraries(test_hl PRIVATE hl hl_langs hl)
target_link_libraries(test_langs PRIVATE hl hl_langs hl)
target_link_libraries(test_sess PRIVATE sess)
target_link_libraries(hlcat PRIVATE hl hl_langs hl)
target_link_libraries(test_ed PRIVATE ed sess hl hl_langs hl)

enable_testing()
add_test(NAME test_hl COMMAND test_hl)
add_test(NAME test_langs COMMAND test_langs ${CMAKE_CURRENT_SOURCE_DIR}/test/samples)
add_test(NAME test_sess COMMAND test_sess ${CMAKE_CURRENT_BINARY_DIR}/sess_test.tmp)
add_test(NAME test_ed_work COMMAND test_ed ${CMAKE_CURRENT_BINARY_DIR}/ed_test.tmp work)
add_test(NAME test_ed_back COMMAND test_ed ${CMAKE_CURRENT_BINARY_DIR}/ed_test.tmp back)
set_tests_properties(test_ed_back PROPERTIES DEPENDS test_ed_work)

if(CMAKE_NM AND NOT MSVC)
    add_test(NAME freestanding_hl COMMAND ${CMAKE_COMMAND} -DNM=${CMAKE_NM} -DPREFIX=hl -DFILES=$<TARGET_FILE:hl>|$<TARGET_FILE:hl_langs> -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/check_freestanding.cmake)
    add_test(NAME freestanding_sess COMMAND ${CMAKE_COMMAND} -DNM=${CMAKE_NM} -DPREFIX=sess -DFILES=$<TARGET_FILE:sess> -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/check_freestanding.cmake)
    add_test(NAME freestanding_ed COMMAND ${CMAKE_COMMAND} -DNM=${CMAKE_NM} "-DPREFIX=(ed|hl|sess)" -DFILES=$<TARGET_FILE:ed> -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/check_freestanding.cmake)
endif()

option(TABPAD_FETCH_SDL3 "download and build SDL3 when it is not installed" ON)
set(TABPAD_SDL3_VERSION 3.4.18)

find_package(SDL3 CONFIG QUIET)

if(NOT SDL3_FOUND AND TABPAD_FETCH_SDL3)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-${TABPAD_SDL3_VERSION}/SDL3-${TABPAD_SDL3_VERSION}.tar.gz)
    FetchContent_MakeAvailable(SDL3)
endif()

if(TARGET SDL3::SDL3)
    add_library(stb_sdl3 OBJECT port/sdl3/stb_sdl3.c)
    target_compile_options(stb_sdl3 PRIVATE ${QUIET})
    target_include_directories(stb_sdl3 PRIVATE vendor/stb)

    set(F ${CMAKE_CURRENT_SOURCE_DIR}/vendor/dejavu/DejaVuSansMono.ttf)
    set(T ${CMAKE_CURRENT_SOURCE_DIR}/translations)
    add_custom_command(OUTPUT font_data.c COMMAND embed -b sdl3_font_data font_data.c ${F} DEPENDS embed ${F})
    add_custom_command(OUTPUT tongue_zh_cn.c COMMAND embed -b sdl3_tongue_zh_cn tongue_zh_cn.c ${T}/zh-CN.txt DEPENDS embed ${T}/zh-CN.txt)
    add_custom_command(OUTPUT tongue_template.c COMMAND embed -b sdl3_tongue_template tongue_template.c ${T}/english.template DEPENDS embed ${T}/english.template)
    set(CARRIED ${CMAKE_CURRENT_BINARY_DIR}/font_data.c ${CMAKE_CURRENT_BINARY_DIR}/tongue_zh_cn.c ${CMAKE_CURRENT_BINARY_DIR}/tongue_template.c)

    add_executable(tabpad WIN32 port/sdl3/main_sdl3.c port/sdl3/draw_sdl3.c port/sdl3/plug_sdl3.c port/sdl3/tongue_sdl3.c ${CARRIED} ${DESK} $<TARGET_OBJECTS:stb_sdl3> ${EDITOR})
    set_target_properties(tabpad PROPERTIES C_STANDARD 99 C_EXTENSIONS OFF)
    target_include_directories(tabpad PRIVATE include lang port port/hosted port/sdl3 vendor/stb)
    target_link_libraries(tabpad PRIVATE ed sess hl hl_langs hl SDL3::SDL3)

    if(NOT MSVC)
        target_link_libraries(tabpad PRIVATE m)
    endif()

    get_target_property(sdl_type SDL3::SDL3 TYPE)

    if(WIN32 AND sdl_type STREQUAL "SHARED_LIBRARY")
        add_custom_command(TARGET tabpad POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:tabpad>)
    endif()
endif()
